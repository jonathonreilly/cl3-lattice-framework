def _read_bytes(path: str) -> bytes:
    return (ROOT / path).read_bytes()


def _read_text(path: str) -> str:
    return _read_bytes(path).decode("utf-8")


def norm(text: str) -> str:
    """Collapse every whitespace run to a single space."""
    return re.sub(r"\s+", " ", text).strip()


def string_constants(path: str) -> list[str]:
    """Every string constant in a Python file, implicit concatenation joined."""
    tree = ast.parse(_read_text(path))
    out: list[str] = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Constant) and isinstance(node.value, str):
            out.append(node.value)
    return out


def q(value: Fraction) -> str:
    """Exact rational rendered as text; no float ever touches a certificate."""
    return f"{value.numerator}/{value.denominator}"


def digest(payload: object) -> str:
    return sha256(
        json.dumps(payload, sort_keys=True, default=str).encode("utf-8")
    ).hexdigest()


def v2(value: Fraction) -> int | None:
    """2-adic valuation of a nonzero rational; None at zero."""
    if value == 0:
        return None
    n, d, e = abs(value.numerator), value.denominator, 0
    while n % 2 == 0:
        n //= 2
        e += 1
    while d % 2 == 0:
        d //= 2
        e -= 1
    return e


def v3(value: Fraction) -> int | None:
    if value == 0:
        return None
    n, d, e = abs(value.numerator), value.denominator, 0
    while n % 3 == 0:
        n //= 3
        e += 1
    while d % 3 == 0:
        d //= 3
        e -= 1
    return e


def rank_exact(rows: list[list[Fraction]]) -> int:
    """Exact rational rank by Gaussian elimination.  No floating point."""
    matrix = [list(row) for row in rows]
    if not matrix:
        return 0
    width = len(matrix[0])
    rank = 0
    for col in range(width):
        pivot = None
        for r in range(rank, len(matrix)):
            if matrix[r][col] != 0:
                pivot = r
                break
        if pivot is None:
            continue
        matrix[rank], matrix[pivot] = matrix[pivot], matrix[rank]
        head = matrix[rank][col]
        matrix[rank] = [x / head for x in matrix[rank]]
        for r in range(len(matrix)):
            if r != rank and matrix[r][col] != 0:
                factor = matrix[r][col]
                matrix[r] = [
                    a - factor * b for a, b in zip(matrix[r], matrix[rank])
                ]
        rank += 1
    return rank


def nullspace_dimension(rows: list[list[Fraction]], width: int) -> int:
    return width - rank_exact(rows)

def signed_permutation_matrices() -> list[tuple[tuple[int, ...], ...]]:
    """All 48 signed permutation matrices on Z^3."""
    out = []
    for perm in product(range(3), repeat=3):
        if len(set(perm)) != 3:
            continue
        for signs in product((1, -1), repeat=3):
            rows = []
            for i in range(3):
                row = [0, 0, 0]
                row[perm[i]] = signs[i]
                rows.append(tuple(row))
            out.append(tuple(rows))
    return out


def det3(m) -> int:
    return (
        m[0][0] * (m[1][1] * m[2][2] - m[1][2] * m[2][1])
        - m[0][1] * (m[1][0] * m[2][2] - m[1][2] * m[2][0])
        + m[0][2] * (m[1][0] * m[2][1] - m[1][1] * m[2][0])
    )


def matmul(a, b):
    return tuple(
        tuple(sum(a[i][k] * b[k][j] for k in range(3)) for j in range(3))
        for i in range(3)
    )


def apply(m, v):
    return tuple(sum(m[i][j] * v[j] for j in range(3)) for i in range(3))


IDENTITY3 = ((1, 0, 0), (0, 1, 0), (0, 0, 1))


def element_order(m) -> int:
    cur, k = m, 1
    while cur != IDENTITY3:
        cur = matmul(cur, m)
        k += 1
        if k > 12:                                        # pragma: no cover
            raise AssertionError("not a finite rotation")
    return k

def orbits_of(m, points) -> list[tuple]:
    seen, out = set(), []
    for p in points:
        if p in seen:
            continue
        orbit, cur = [], p
        while cur not in orbit:
            orbit.append(cur)
            cur = apply(m, cur)
        for x in orbit:
            seen.add(x)
        out.append(tuple(orbit))
    return out

def isotype_pair_over_Q(n: int) -> tuple[int, int]:
    """(invariant dim, complement dim) for the cyclic permutation rep on n
    points, computed by exact rational elimination."""
    # Invariant vectors a satisfy a_i = a_{sigma(i)} for the n-cycle sigma.
    rows = []
    for i in range(n):
        row = [Fraction(0)] * n
        row[i] += 1
        row[(i + 1) % n] -= 1
        rows.append(row)
    invariant = nullspace_dimension(rows, n)
    return invariant, n - invariant

def cycle318_mediator_weight_by_ast() -> dict:
    """Recover Cycle-318's mediator weight from the source, by AST."""
    tree = ast.parse(_read_text(AUDIT_INPUT_PATHS[6]))
    found = []
    for node in ast.walk(tree):
        if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            continue
        args = node.args
        names = [a.arg for a in args.args]
        defaults = list(args.defaults)
        offset = len(names) - len(defaults)
        for i, name in enumerate(names):
            if name != "mediator_weight":
                continue
            if i < offset:
                continue
            default = defaults[i - offset]
            if isinstance(default, ast.Constant):
                found.append({
                    "function": node.name,
                    "line": node.lineno,
                    "argument": name,
                    "default_value": default.value,
                    "is_a_default_argument": True,
                })
    return {"occurrences": found}

# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(certs: dict) -> str:
    out = ["CYCLE 883 -- SL1: DOES THE RECORD CARRY THE WEIGHT PAIR (1, 2)?", ""]
    for label in LABELS:
        cert = certs[label]
        out.append(f"[{'PASS' if cert['pass'] else 'FAIL'}] {label}")
        finding = cert.get("finding")
        if finding:
            out.append(f"    finding: {finding}")
        out.append("")
    out.append(json.dumps(certs, indent=2, sort_keys=True, default=str))
    return "\n".join(out) + "\n"


def build_science() -> dict:
    sweep = sweep_tree()
    pins = pins_certificate()
    clauses = axiom_clause_certificate()
    inventory = multiplicity_inventory_certificate()
    candidates = candidate_sweep_certificate(sweep)

    t6 = bridge_back_t6_certificate()

    binding = binding_price_certificate()
    echoes = echo_ledger_certificate()
    terminal = terminal_lemma_certificate(isotype, binding, t7)
    outcome = outcome_certificate(sweep, isotype, t6)
    science = {
        "A_PINS": pins,
        "B_AXIOM_CLAUSES": clauses,

        "D_CANDIDATE_SWEEP": candidates,
